from pyscript import document # Données initiales magasin = [ {'nom': 'Excalibur', 'prix': 2500, 'type': 'arme'}, {'nom': 'Potion de vie', 'prix': 500, 'type': 'potion'}, {'nom': 'Plastron de Dragon de Jade', 'prix': 850, 'type': 'armure'} ] def ecrire_console(texte): console_element = document.querySelector("#console-log") console_element.innerText = texte # 1. Calculer valeur totale def btn_valeur_totale(event): total = 0 for objet in magasin: total += objet['prix'] ecrire_console(f"💰 Le montant total du magasin est de {total} pièces d'or.") # 2. Trier par prix def btn_trier(event): inventaire_trie = sorted(magasin, key=lambda x: x['prix']) resultat = "📉 Liste triée par prix :\n" for o in inventaire_trie: resultat += f"- {o['nom']} ({o['prix']} po) [{o['type']}]\n" ecrire_console(resultat) # 3. Filtrer par type def btn_filtrer(event): type_rech = document.querySelector("#input_type").value.strip() trouve = False resultat = f"🔍 Résultats pour le type '{type_rech}' :\n" for objet in magasin: if objet['type'] == type_rech: resultat += f"- {objet['nom']} ({objet['prix']} po)\n" trouve = True if not trouve: resultat += f"Désolé, nous n'avons pas ce type d'objets !" ecrire_console(resultat) # 4. Ajouter un objet def btn_ajouter(event): nom = document.querySelector("#input_type").value # ou champs dédiés nom_input = document.querySelector("#ajout_nom").value.strip() prix_str = document.querySelector("#ajout_prix").value.strip() type_obj = document.querySelector("#ajout_type").value.strip() if nom_input and prix_str and type_obj: nouveau_prix = int(prix_str) magasin.append({'nom': nom_input, 'prix': nouveau_prix, 'type': type_obj}) ecrire_console(f"✅ Succès ! '{nom_input}' a été ajouté au magasin.") else: ecrire_console("❌ Erreur : Veuillez remplir tous les champs pour l'ajout.") # 5. Supprimer un objet (La nouveauté !) def btn_supprimer(event): nom_a_supprimer = document.querySelector("#suppr_nom").value.strip() trouve = False # On parcourt la liste à l'envers ou on filtre, # ou on utilise la logique classique de suppression par boucle : for objet in magasin: if objet['nom'].lower() == nom_a_supprimer.lower(): magasin.remove(objet) trouve = True break if trouve: ecrire_console(f"🗑️ L'objet '{nom_a_supprimer}' a été supprimé avec succès !") else: ecrire_console(f"❌ Introuvable : Aucun objet nommé '{nom_a_supprimer}' dans le magasin.")
